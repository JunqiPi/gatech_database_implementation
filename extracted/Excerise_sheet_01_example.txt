4. Consider the following function:
#include <memory>
#include <stdexcept>
#include <vector>
 
constexpr size_t PAGE_SIZE = 4096;
 
void bulkLoad(int n) {
  std::vector<std::unique_ptr<char[]>> pages;
 
  for (int i = 1; i <= n; ++i) {
    pages.push_back(std::make_unique<char[]>(PAGE_SIZE));
    if (i == n / 2) {  // simulating failure halfway
      throw std::runtime_error("Disk full during bulk load!");
    }
  }
}
 
If bulkLoad(10) is called, how many pages are freed during exception unwinding?
 
A) 0
 
B) 5
 
C) 10
